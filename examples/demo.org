#+begin_src python :session cw-demo :results value file graphics :file demo-plot.png :var amp=(cw-value "amp") :var freq=(cw-value "freq") :var grid=(cw-value "grid") :var style=(cw-value "style") :var seed=(cw-value "noise")
import matplotlib; matplotlib.use("Agg")
import matplotlib.pyplot as plt, numpy as np
rng = np.random.default_rng(seed)
x = np.linspace(0, 2*np.pi, 400)
fig, ax = plt.subplots(figsize=(6, 3))
ax.plot(x, amp*np.sin(freq*x) + 0.05*rng.standard_normal(x.size), style)
ax.set_ylim(-5.5, 5.5); ax.grid(bool(grid))
plt.close(fig)
